cmake_minimum_required(VERSION 3.15)
project(SlmBackendA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 构建设置：生成后端库、独立测试和控制台示例。
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(slm_backend_a core/Fft.cpp core/Asm.cpp core/Metrics.cpp core/Axial.cpp)
target_include_directories(slm_backend_a PUBLIC core)
add_executable(CoreSmokeTests tests/CoreSmokeTests.cpp)
target_link_libraries(CoreSmokeTests PRIVATE slm_backend_a)
add_executable(BackendADemo examples/BackendADemo.cpp)
target_link_libraries(BackendADemo PRIVATE slm_backend_a)
enable_testing()
add_test(NAME BackendACoreTests COMMAND CoreSmokeTests)
